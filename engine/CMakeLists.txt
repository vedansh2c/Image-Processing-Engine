cmake_minimum_required(VERSION 3.20)
project(ImageProcessingEngine VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(PF_BUILD_TESTS "Build GoogleTest tests" ON)
option(PF_BUILD_PYTHON "Build pybind11 extension" ON)
option(PF_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(PF_THREAD_SANITIZE "Enable ThreadSanitizer separately" OFF)
if(PF_SANITIZE AND PF_THREAD_SANITIZE)
  message(FATAL_ERROR "ASan and TSan must run in separate builds")
endif()
find_package(Threads REQUIRED)
add_library(image_processing_engine src/core/image.cpp src/filters/filters.cpp src/parallel/thread_pool.cpp src/optimization/invert.cpp src/benchmark/benchmark.cpp)
target_include_directories(image_processing_engine PUBLIC include)
target_include_directories(image_processing_engine SYSTEM PRIVATE third_party)
target_link_libraries(image_processing_engine PUBLIC Threads::Threads)
target_compile_options(image_processing_engine PRIVATE -Wall -Wextra -Wpedantic)
if(PF_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(image_processing_engine PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(PF_THREAD_SANITIZE)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  target_compile_options(image_processing_engine PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
if(PF_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir OUTPUT_VARIABLE PYBIND_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
  list(APPEND CMAKE_PREFIX_PATH "${PYBIND_DIR}")
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(image_processing_engine_native bindings/module.cpp)
  target_link_libraries(image_processing_engine_native PRIVATE image_processing_engine)
  install(TARGETS image_processing_engine_native LIBRARY DESTINATION .)
endif()
add_executable(image_processing_engine_bench src/benchmark/main.cpp)
target_link_libraries(image_processing_engine_bench PRIVATE image_processing_engine)
if(PF_BUILD_TESTS)
  include(CTest)
  include(FetchContent)
  FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(googletest)
  add_executable(engine_tests tests/engine_test.cpp)
  target_link_libraries(engine_tests PRIVATE image_processing_engine GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(engine_tests)
endif()
